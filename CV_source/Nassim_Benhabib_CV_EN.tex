\cvsection{Engineering \& research experience}
\position{UAVIA}{Autonomous Navigation \& Control Engineer}{Oct. 2022 -- Mar. 2026}
\begin{itemize}\small
\item \textbf{Constrained predictive control.} Developed a C++ MPC controller formulated as a quadratic program with qpOASES, accounting for velocity, acceleration, braking and yaw constraints.
\item \textbf{Motion planning.} Combined global Dijkstra search, local safety corridors and MPC to generate feasible trajectories.
\item \textbf{Guidance and precision landing.} Designed RTK/IMU-based approach logic with an evolving capture zone, phased descent and contact detection.
\item \textbf{Safety and degraded modes.} Implemented fallback behaviors and geofencing accounting for restricted areas, wind, position uncertainty and projected fall trajectories.
\item \textbf{Mission-specific navigation.} Developed terrain-following with altitude adaptation to terrain, and navigation for detecting and covering CO$_2$ and CH$_4$ emission plumes.
\item \textbf{Integration and validation.} Integrated autonomy services with DJI, Parrot, PX4 and ArduPilot; validated them through simulation, telemetry analysis, tuning and flight tests.
\end{itemize}
\vspace{2pt}

\position{Inria Bordeaux}{Doctoral Researcher in Robotics}{Nov. 2018 -- 2022}
\begin{itemize}\small
\item \textbf{Modeling and optimization.} Built a dynamic model of human--robot--environment interaction; formulated inverse dynamics and force estimation as quadratic programs under contact constraints and physical limits.
\item \textbf{System identification.} Estimated parameters from measured trajectories and forces; assessed prediction errors on independent experimental trials.
\item \textbf{Simulation and experimental validation.} Developed a numerical simulator and an instrumented robotic test platform; compared simulated and measured motion and forces under nominal and perturbed conditions.
\item \textbf{Human--robot assistance control.} Implemented adaptive assistance on the experimental platform using predictive control, quadratic optimization and physical constraints.
\end{itemize}
